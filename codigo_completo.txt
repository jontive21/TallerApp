using Microsoft.IdentityModel.Tokens;

using System.Security.Claims;
using System.Text;
using TallerApp.Web.Data;
using TallerApp.Web.Models;

namespace TallerApp.Web.Controllers.Api
{
    [Route("api/[controller]")]
    [ApiController]
    public class ApiAuthController : ControllerBase
    {
        private readonly IRepositorioUsuario _repoUsuario;

        public ApiAuthController(IRepositorioUsuario repoUsuario)
        {
            _repoUsuario = repoUsuario;
        }

        [HttpPost("login")]
        public IActionResult Login([FromBody] LoginViewModel model)
        {
            // 1. Buscar el usuario
            var usuario = _repoUsuario.ObtenerPorEmail(model.Email);

            // 2. Verificar que exista y la contraseÃ±a sea correcta
            if (usuario == null || !PasswordHelper.VerificarHash(model.Password, usuario.PasswordHash))
            {
                return Unauthorized(new { mensaje = "Credenciales invÃ¡lidas" });
            }

            // 3. Crear los Claims (la informaciÃ³n que irÃ¡ dentro del token)
            var claims = new[]
            {
                new Claim(ClaimTypes.NameIdentifier, usuario.IdUsuario.ToString()),
                new Claim(ClaimTypes.Name, usuario.NombreCompleto),
                new Claim(ClaimTypes.Role, usuario.Rol)
            };

            // 4. Configurar la clave secreta (debe ser la misma que en Program.cs)
            var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes("TuClaveSecretaMuyLargaYSeguraDeAlMenos32Caracteres123!"));
            var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

            // 5. Generar el Token
            var token = new JwtSecurityToken(
                issuer: "TallerAppIssuer",
                audience: "TallerAppAudience",
                claims: claims,
                expires: DateTime.Now.AddHours(2), // El token dura 2 horas
                signingCredentials: creds
            );

            // 6. Devolver el token al cliente
            return Ok(new
            {
                token = new JwtSecurityTokenHandler().WriteToken(token),
                usuario = usuario.NombreCompleto,
                rol = usuario.Rol
            });
        }
    }

using System.Security.Claims;
using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using TallerApp.Web.Models;
using TallerApp.Web.Data;

namespace TallerApp.Web.Controllers
{
    public class AccountController : Controller
    {
        private readonly IRepositorioUsuario _repoUsuario;
        private readonly IWebHostEnvironment _entorno;

        private static readonly string[] ExtensionesPermitidas = { ".jpg", ".jpeg", ".png", ".webp", ".gif" };
        private const long TamanioMaximoBytes = 2 * 1024 * 1024; // 2 MB

        public AccountController(IRepositorioUsuario repoUsuario, IWebHostEnvironment entorno)
        {
            _repoUsuario = repoUsuario;
            _entorno = entorno;
        }

        // ===================== LOGIN / LOGOUT =====================
        [HttpGet]
        public IActionResult Login(string? returnUrl = null)
        {
            ViewData["ReturnUrl"] = returnUrl;
            return View();
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Login(LoginViewModel model, string? returnUrl = null)
        {
            ViewData["ReturnUrl"] = returnUrl;

            if (ModelState.IsValid)
            {
                var usuario = _repoUsuario.ObtenerPorEmail(model.Email);

                if (usuario != null && PasswordHelper.VerificarHash(model.Password, usuario.PasswordHash))
                {
                    var claims = new List<Claim>
                    {
                        new Claim(ClaimTypes.Name, usuario.NombreCompleto),
                        new Claim(ClaimTypes.NameIdentifier, usuario.IdUsuario.ToString()),
                        new Claim(ClaimTypes.Role, usuario.Rol)
                    };

                    var claimsIdentity = new ClaimsIdentity(claims, CookieAuthenticationDefaults.AuthenticationScheme);
                    var claimsPrincipal = new ClaimsPrincipal(claimsIdentity);

                    await HttpContext.SignInAsync(CookieAuthenticationDefaults.AuthenticationScheme, claimsPrincipal);

                    return LocalRedirect(returnUrl ?? "/");
                }

                ModelState.AddModelError(string.Empty, "Usuario o contraseña incorrectos.");
            }

            return View(model);
        }

        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Logout()
        {
            await HttpContext.SignOutAsync(CookieAuthenticationDefaults.AuthenticationScheme);
            return RedirectToAction("Index", "Home");
        }

        // ===================== PERFIL PROPIO =====================
        [Authorize]
        public IActionResult Perfil()
        {
            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();
            return View(usuario);
        }

        // ----- Editar datos personales -----
        [Authorize]
        [HttpGet]
        public IActionResult EditarDatos()
        {
            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();

            var model = new EditarDatosViewModel
            {
                NombreCompleto = usuario.NombreCompleto,
                Email = usuario.Email
            };
            return View(model);
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> EditarDatos(EditarDatosViewModel model)
        {
            if (!ModelState.IsValid) return View(model);

            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();

            try
            {
                usuario.NombreCompleto = model.NombreCompleto;
                usuario.Email = model.Email;
                _repoUsuario.Modificacion(usuario);

                await RefrescarSesionAsync(usuario);

                TempData["Mensaje"] = "Datos personales actualizados correctamente.";
                return RedirectToAction(nameof(Perfil));
            }
            catch
            {
                ViewBag.Error = "No se pudieron actualizar los datos (posible email duplicado).";
                return View(model);
            }
        }

        // ----- Cambiar contraseña -----
        [Authorize]
        [HttpGet]
        public IActionResult CambiarClave()

            return View();

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult CambiarClave(CambiarClaveViewModel model)

            if (!ModelState.IsValid) return View(model);

            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();

            if (!PasswordHelper.VerificarHash(model.ClaveActual, usuario.PasswordHash))
            {
                ModelState.AddModelError("ClaveActual", "La contraseña actual es incorrecta.");
                return View(model);
            }

            try
            {
                usuario.PasswordHash = PasswordHelper.GenerarHash(model.ClaveNueva);
                _repoUsuario.Modificacion(usuario);

                TempData["Mensaje"] = "Contraseña cambiada correctamente.";
                return RedirectToAction(nameof(Perfil));
            }
            catch
            {
                ViewBag.Error = "No se pudo cambiar la contraseña.";
                return View(model);
            }
        }

        // ----- Cambiar Avatar -----
        [Authorize]
        [HttpGet]
        public IActionResult CambiarAvatar()
        {
            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();
            return View(usuario);
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult CambiarAvatar(IFormFile? AvatarFile)
        {
            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();

            if (AvatarFile == null || AvatarFile.Length == 0)
            {
                ViewBag.Error = "Debés seleccionar una imagen.";
                return View(usuario);
            }

            try
            {
                var ruta = GuardarAvatar(AvatarFile);
                if (ruta == null)
                {
                    ViewBag.Error = "Formato o tamaño de imagen no válido (máx 2MB, jpg/png/webp/gif).";
                    return View(usuario);
                }

                EliminarArchivoSiExiste(usuario.Avatar);
                usuario.Avatar = ruta;
                _repoUsuario.Modificacion(usuario);

                TempData["Mensaje"] = "Avatar actualizado correctamente.";
                return RedirectToAction(nameof(Perfil));
            }
            catch (Exception ex)
            {
                ViewBag.Error = "Error al subir el avatar: " + ex.Message;
                return View(usuario);
            }
        }

        // ----- Eliminar Avatar -----
        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public IActionResult EliminarAvatar()
        {
            var usuario = ObtenerUsuarioActual();
            if (usuario == null) return NotFound();

            try
            {
                EliminarArchivoSiExiste(usuario.Avatar);
                usuario.Avatar = null;
                _repoUsuario.Modificacion(usuario);

                TempData["Mensaje"] = "Avatar eliminado correctamente.";
            }
            catch
            {
                TempData["Error"] = "No se pudo eliminar el avatar.";
            }

            return RedirectToAction(nameof(Perfil));
        }

        // ===================== HELPERS =====================
        private async Task RefrescarSesionAsync(Usuario usuario)
        {
            var claims = new List<Claim>
            {
                new Claim(ClaimTypes.Name, usuario.NombreCompleto),
                new Claim(ClaimTypes.NameIdentifier, usuario.IdUsuario.ToString()),
                new Claim(ClaimTypes.Role, usuario.Rol)
            };
            var identity = new ClaimsIdentity(claims, CookieAuthenticationDefaults.AuthenticationScheme);
            await HttpContext.SignInAsync(CookieAuthenticationDefaults.AuthenticationScheme, new ClaimsPrincipal(identity));
        }

        private Usuario? ObtenerUsuarioActual()
        {
            var idClaim = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
            if (!int.TryParse(idClaim, out int id)) return null;
            return _repoUsuario.ObtenerPorId(id);
        }

        private string? GuardarAvatar(IFormFile archivo)
        {
            var extension = Path.GetExtension(archivo.FileName).ToLowerInvariant();
            if (!ExtensionesPermitidas.Contains(extension) || archivo.Length > TamanioMaximoBytes)
                return null;

            var nombreArchivo = $"avatar_{Guid.NewGuid()}{extension}";
            var carpeta = Path.Combine(_entorno.WebRootPath, "img", "avatars");

            if (!Directory.Exists(carpeta))
                Directory.CreateDirectory(carpeta);

            var rutaFisica = Path.Combine(carpeta, nombreArchivo);
            using (var stream = new FileStream(rutaFisica, FileMode.Create))
            {
                archivo.CopyTo(stream);
            }

            return $"/img/avatars/{nombreArchivo}";
        }

        private void EliminarArchivoSiExiste(string? rutaRelativa)
        {
            if (string.IsNullOrEmpty(rutaRelativa)) return;
            if (!rutaRelativa.StartsWith("/img/avatars/", StringComparison.OrdinalIgnoreCase)) return;

            var rutaFisica = Path.Combine(_entorno.WebRootPath, rutaRelativa.TrimStart('/'));
            if (System.IO.File.Exists(rutaFisica))
            {
                System.IO.File.Delete(rutaFisica);
            }
        }
    }
}

                    {

        {

            {
                sql += " WHERE NombreCompleto LIKE @busqueda OR DNI LIKE @busqueda OR Email LIKE @busqueda";

            sql += " LIMIT @limite OFFSET @offset";
            
            using var cmd = new MySqlCommand(sql, conexion);
            if (!string.IsNullOrEmpty(busqueda))
            {
                cmd.Parameters.AddWithValue("@busqueda", $"%{busqueda}%");
            }
            cmd.Parameters.AddWithValue("@limite", registrosPorPagina);
            cmd.Parameters.AddWithValue("@offset", (pagina - 1) * registrosPorPagina);
            
            using var reader = cmd.ExecuteReader();
            while (reader.Read())

using Microsoft.Extensions.Configuration;
using System.Collections.Generic;
using MySqlConnector;
using TallerApp.Web.Models;

namespace TallerApp.Web.Data
{
    public class RepositorioOrdenTrabajo : RepositorioBase, IRepositorioOrdenTrabajo
    {
        public RepositorioOrdenTrabajo(IConfiguration configuration) : base(configuration) { }

        public int CrearOrdenConRepuestos(OrdenTrabajo orden, List<OrdenRepuesto> detalles)
        {
            using var conexion = ObtenerConexion();
            conexion.Open();
            
            // Iniciamos una transacción para garantizar integridad
            using var transaction = conexion.BeginTransaction();
            
            try
            {
                // 1. Insertar la Orden de Trabajo
                string sqlOrden = @"INSERT INTO ordenestrabajo 
                    (VehiculoId, MecanicoId, FechaIngreso, FechaEntregaEstimada, Estado, DescripcionProblema, Observaciones, IdUsuarioCreacion) 
                    VALUES (@vehiculo, @mecanico, @fechaIngreso, @fechaEntrega, @estado, @descripcion, @observaciones, @usuario)";
                
                using var cmdOrden = new MySqlCommand(sqlOrden, conexion, transaction);
                cmdOrden.Parameters.AddWithValue("@vehiculo", orden.VehiculoId);
                cmdOrden.Parameters.AddWithValue("@mecanico", orden.MecanicoId);
                cmdOrden.Parameters.AddWithValue("@fechaIngreso", orden.FechaIngreso);
                cmdOrden.Parameters.AddWithValue("@fechaEntrega", (object)orden.FechaEntregaEstimada ?? DBNull.Value);
                cmdOrden.Parameters.AddWithValue("@estado", orden.Estado);
                cmdOrden.Parameters.AddWithValue("@descripcion", orden.DescripcionProblema);
                cmdOrden.Parameters.AddWithValue("@observaciones", (object)orden.Observaciones ?? DBNull.Value);
                cmdOrden.Parameters.AddWithValue("@usuario", orden.IdUsuarioCreacion);
                
                cmdOrden.ExecuteNonQuery();
                
                // Obtener el ID generado
                cmdOrden.CommandText = "SELECT LAST_INSERT_ID();";
                int nuevaOrdenId = Convert.ToInt32(cmdOrden.ExecuteScalar());

                // 2. Insertar los detalles (Repuestos) y descontar stock
                string sqlDetalle = @"INSERT INTO OrdenRepuesto (OrdenId, RepuestoId, Cantidad, PrecioUnitario) 
                                      VALUES (@ordenId, @repuestoId, @cantidad, @precio)";
                
                string sqlStock = "UPDATE Repuestos SET Stock = Stock - @cantidad WHERE IdRepuesto = @repuestoId";

                foreach (var detalle in detalles)
                {
                    using var cmdDetalle = new MySqlCommand(sqlDetalle, conexion, transaction);
                    cmdDetalle.Parameters.AddWithValue("@ordenId", nuevaOrdenId);
                    cmdDetalle.Parameters.AddWithValue("@repuestoId", detalle.RepuestoId);
                    cmdDetalle.Parameters.AddWithValue("@cantidad", detalle.Cantidad);
                    cmdDetalle.Parameters.AddWithValue("@precio", detalle.PrecioUnitario);
                    cmdDetalle.ExecuteNonQuery();

                    // Descontar del stock
                    using var cmdStock = new MySqlCommand(sqlStock, conexion, transaction);
                    cmdStock.Parameters.AddWithValue("@cantidad", detalle.Cantidad);
                    cmdStock.Parameters.AddWithValue("@repuestoId", detalle.RepuestoId);
                    cmdStock.ExecuteNonQuery();
                }

                // 3. Confirmar la transacción
                transaction.Commit();
                return nuevaOrdenId;
            }
            catch
            {
                // Si algo falla, deshacer todo
                transaction.Rollback();
                throw;
            }
        }
    }
}

using System.Security.Cryptography;

namespace TallerApp.Web.Models
{
    /// <summary>
    /// Genera y verifica hashes de contraseÃ±a usando PBKDF2 (estÃ¡ndar, seguro).
    /// No guarda la contraseÃ±a en texto plano, solo el hash + la sal.
    /// </summary>
    public static class PasswordHelper
    {
        private const int Iteraciones = 100_000;
        private const int TamanioSal = 16;   // bytes
        private const int TamanioHash = 32;  // bytes

        // Genera un hash nuevo a partir de una contraseÃ±a en texto plano.
        // Formato guardado: "iteraciones.saltEnBase64.hashEnBase64"
        public static string GenerarHash(string password)
        {
            byte[] sal = RandomNumberGenerator.GetBytes(TamanioSal);
            byte[] hash = Rfc2898DeriveBytes.Pbkdf2(password, sal, Iteraciones, HashAlgorithmName.SHA256, TamanioHash);

            return $"{Iteraciones}.{Convert.ToBase64String(sal)}.{Convert.ToBase64String(hash)}";
        }

        // Verifica que una contraseÃ±a en texto plano coincida con un hash almacenado.
        public static bool VerificarHash(string password, string hashAlmacenado)
        {
            var partes = hashAlmacenado.Split('.', 3);
            if (partes.Length != 3) return false;

            if (!int.TryParse(partes[0], out int iteraciones)) return false;
            byte[] sal, hashGuardado;
            try
            {
                sal = Convert.FromBase64String(partes[1]);
                hashGuardado = Convert.FromBase64String(partes[2]);
            }
            catch (FormatException)
            {
                return false; // hash corrupto o en formato viejo
            }

            byte[] hashIngresado = Rfc2898DeriveBytes.Pbkdf2(password, sal, iteraciones, HashAlgorithmName.SHA256, hashGuardado.Length);

            return CryptographicOperations.FixedTimeEquals(hashIngresado, hashGuardado);
        }
    }
}

    <!-- Vue 3 CDN -->
    <script src="https://unpkg.com/vue@3/dist/vue.global.js"></script>

    <script>
        const { createApp } = Vue;

        createApp({
            data() {
                return {
                    // Datos principales
                    clienteId: '',
                    vehiculoId: '',
                    mecanicoId: '',
                    fechaIngreso: new Date().toISOString().split('T')[0],
                    descripcionProblema: '',
                    observaciones: '',

                    // BÃºsqueda asÃ­ncrona
                    busquedaCliente: '',
                    busquedaMecanico: '',

                    // Listas
                    clientes: [],
                    vehiculos: [],
                    mecanicos: [],
                    repuestosFiltrados: [],
                    items: [],          // repuestos agregados a la orden

                    // Auxiliares
                    busquedaRepuesto: '',
                    repuestoSeleccionado: null,
                    total: 0,
                    guardando: false,
                    mensaje: '',
                    mensajeExito: false
                }
            },
            mounted() {
                // Ya no cargamos todos los clientes y mecÃ¡nicos al inicio
                // Se cargarÃ¡n bajo demanda con la bÃºsqueda
            },
            methods: {
                async buscarClientes() {
                    if (this.busquedaCliente.length < 2) { this.clientes = []; return; }
                    const res = await fetch(`/Ordenes/GetClientes?busqueda=${encodeURIComponent(this.busquedaCliente)}`);
                    this.clientes = await res.json();
                },
                async buscarMecanicos() {
                    if (this.busquedaMecanico.length < 2) { this.mecanicos = []; return; }
                    const res = await fetch(`/Ordenes/GetMecanicos?busqueda=${encodeURIComponent(this.busquedaMecanico)}`);
                    this.mecanicos = await res.json();
                },
                async cargarVehiculos() {
                    this.vehiculoId = '';
                    this.vehiculos = [];
                    if (!this.clienteId) return;

                    const res = await fetch(`/Ordenes/GetVehiculosPorCliente?clienteId=${this.clienteId}`);
                    this.vehiculos = await res.json();
                },
                async buscarRepuestos() {
                    if (this.busquedaRepuesto.length < 2) {
                        this.repuestosFiltrados = [];
                        return;
                    }
                    const res = await fetch(`/Ordenes/GetRepuestos?busqueda=${encodeURIComponent(this.busquedaRepuesto)}`);
                    this.repuestosFiltrados = await res.json();
                },
                agregarRepuesto() {
                    if (!this.repuestoSeleccionado) return;

                    // Evitar duplicados
                    const existe = this.items.find(i => i.repuestoId === this.repuestoSeleccionado.idRepuesto);
                    if (existe) {
                        existe.cantidad++;
                    } else {
                        this.items.push({
                            repuestoId: this.repuestoSeleccionado.idRepuesto,
                            nombre: this.repuestoSeleccionado.nombre,
                            codigo: this.repuestoSeleccionado.codigo,
                            cantidad: 1,
                            precioUnitario: this.repuestoSeleccionado.precio
                        });
                    }

                    this.repuestoSeleccionado = null;
                    this.busquedaRepuesto = '';
                    this.repuestosFiltrados = [];
                    this.calcularTotales();
                },
                eliminarItem(index) {
                    this.items.splice(index, 1);
                    this.calcularTotales();
                },
                calcularTotales() {
                    this.total = this.items.reduce((sum, item) => {
                        return sum + (item.cantidad * item.precioUnitario);
                    }, 0);
                },
                async guardarOrden() {
                    // Validaciones bÃ¡sicas
                    if (!this.clienteId || !this.vehiculoId || !this.mecanicoId) {
                        this.mensaje = 'Debe seleccionar Cliente, VehÃ­culo y MecÃ¡nico';
                        this.mensajeExito = false;
                        return;
                    }

                    this.guardando = true;
                    this.mensaje = '';

                    const orden = {
                        vehiculoId: parseInt(this.vehiculoId),
                        mecanicoId: parseInt(this.mecanicoId),
                        fechaIngreso: this.fechaIngreso,
                        descripcionProblema: this.descripcionProblema,
                        observaciones: this.observaciones,
                        estado: 'Pendiente',
                        repuestos: this.items.map(i => ({
                            repuestoId: i.repuestoId,
                            cantidad: i.cantidad,
                            precioUnitario: i.precioUnitario
                        }))
                    };

                    try {
                        const res = await fetch('/Ordenes/CrearOrden', {
                            method: 'POST',
                            headers: {
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify(orden)
                        });

                        if (res.ok) {
                            this.mensaje = 'Orden guardada correctamente';
                            this.mensajeExito = true;

                            // Limpiar formulario
                            setTimeout(() => {
                                window.location.href = '/Ordenes';
                            }, 1500);
                        } else {
                            const error = await res.text();
                            this.mensaje = 'Error al guardar: ' + error;
                            this.mensajeExito = false;
                        }
                    } catch (error) {
                        this.mensaje = 'Error de conexiÃ³n: ' + error.message;
                        this.mensajeExito = false;
                    } finally {
                        this.guardando = false;
                    }
                }
            }
        }).mount('#app');
    </script>
}

using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.IdentityModel.Tokens;
using System.Text;
using TallerApp.Web.Data;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllersWithViews();
builder.Services.AddControllers(); // Necesario para la API

// Registrar los repositorios
builder.Services.AddScoped<IRepositorioUsuario, RepositorioUsuario>();
builder.Services.AddScoped<IRepositorioCliente, RepositorioCliente>();
builder.Services.AddScoped<IRepositorioVehiculo, RepositorioVehiculo>();
builder.Services.AddScoped<IRepositorioMecanico, RepositorioMecanico>();
builder.Services.AddScoped<IRepositorioRepuesto, RepositorioRepuesto>();
builder.Services.AddScoped<IRepositorioOrdenTrabajo, RepositorioOrdenTrabajo>();
builder.Services.AddScoped<IRepositorioComprobantePago, RepositorioComprobantePago>();

// 1. Configurar Autorización (Políticas de Roles)
builder.Services.AddAuthorization(options =>
{
    options.AddPolicy("SoloAdmin", policy => policy.RequireRole("Admin"));
});

// 2. Configurar Autenticación: Cookies (Web por defecto) + JWT (API)
builder.Services.AddAuthentication(options =>
{
    // ¡IMPORTANTE! El esquema por defecto es Cookies para que la web funcione normal
    options.DefaultScheme = CookieAuthenticationDefaults.AuthenticationScheme;
    options.DefaultChallengeScheme = CookieAuthenticationDefaults.AuthenticationScheme;
})
.AddCookie(CookieAuthenticationDefaults.AuthenticationScheme, options => {
    options.LoginPath = "/Account/Login";
    options.AccessDeniedPath = "/Account/AccesoDenegado";
})
.AddJwtBearer(JwtBearerDefaults.AuthenticationScheme, options =>
{
    options.TokenValidationParameters = new TokenValidationParameters
    {
        ValidateIssuer = true,
        ValidateAudience = true,
        ValidateLifetime = true,
        ValidateIssuerSigningKey = true,
        ValidIssuer = "TallerAppIssuer",
        ValidAudience = "TallerAppAudience",
        IssuerSigningKey = new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes("TuClaveSecretaMuyLargaYSeguraDeAlMenos32Caracteres123!")
        )
    };
});

var app = builder.Build();

if (!app.Environment.IsDevelopment())
{
    app.UseExceptionHandler("/Home/Error");
    app.UseHsts();
}

app.UseHttpsRedirection();
app.UseStaticFiles();
app.UseRouting();

// ¡El orden es importante!
app.UseAuthentication();
app.UseAuthorization();

app.MapControllerRoute(
    name: "default",
    pattern: "{controller=Home}/{action=Index}/{id?}");

app.Run();
